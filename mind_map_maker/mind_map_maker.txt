// ===================== 新增：持久化核心函数 =====================
// 1. 保存所有数据到localStorage
function saveToLocalStorage() {
  // 收集框的信息
  const boxes = Array.from(document.querySelectorAll('.box')).map(b => ({
    left: b.style.left,
    top: b.style.top,
    content: b.innerHTML,
    id: b.dataset.id,
    fontSize: b.style.fontSize,
    fontWeight: b.style.fontWeight,
    color: b.style.color
  }));
  
  // 收集视图状态（缩放、偏移）
  const viewState = {
    scale,
    offsetX,
    offsetY
  };
  
  // 收集连线
  const savedConnections = [...connections];
  
  // 组合成完整数据并存储
  const canvasData = {
    boxes,
    connections: savedConnections,
    viewState
  };
  
  localStorage.setItem('canvasData', JSON.stringify(canvasData));
}

// 2. 从localStorage加载数据
function loadFromLocalStorage() {
  const savedData = localStorage.getItem('canvasData');
  if (!savedData) return false;
  
  try {
    const { boxes, connections: savedConnections, viewState } = JSON.parse(savedData);
    
    // 清空现有内容
    document.querySelectorAll('.box').forEach(b => b.remove());
    connections = [];
    
    // 恢复视图状态
    if (viewState) {
      scale = viewState.scale || 1;
      offsetX = viewState.offsetX || 0;
      offsetY = viewState.offsetY || 0;
      updateViewport();
    }
    
    // 恢复框
    const boxMap = new Map();
    boxes.forEach(data => {
      const box = document.createElement('div');
      box.className = 'box';
      box.contentEditable = true;
      box.style.left = data.left;
      box.style.top = data.top;
      box.innerHTML = data.content;
      box.style.fontSize = data.fontSize || '15px';
      box.style.fontWeight = data.fontWeight || '400';
      box.style.color = data.color || '#333';
      box.dataset.id = data.id || Date.now().toString();
      viewport.appendChild(box);
      makeDraggable(box);
      boxMap.set(data.id, box);
    });
    
    // 恢复连线
    connections = savedConnections || [];
    redrawAllConnections();
    
    return true;
  } catch (e) {
    console.error('加载本地数据失败：', e);
    return false;
  }
}

// 3. 导出数据为JSON文件
function exportData() {
  const savedData = localStorage.getItem('canvasData');
  if (!savedData) {
    alert('暂无数据可导出！');
    return;
  }
  
  const blob = new Blob([savedData], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `画布数据_${new Date().toLocaleString().replace(/[/: ]/g, '-')}.json`;
  a.click();
  URL.revokeObjectURL(url);
}

// 4. 导入JSON文件恢复数据
function importData(event) {
  const file = event.target.files[0];
  if (!file) return;
  
  if (file.type !== 'application/json' && !file.name.endsWith('.json')) {
    alert('请选择JSON格式的文件！');
    return;
  }
  
  const reader = new FileReader();
  reader.onload = function(e) {
    try {
      const canvasData = JSON.parse(e.target.result);
      localStorage.setItem('canvasData', JSON.stringify(canvasData));
      loadFromLocalStorage();
      alert('数据导入成功！');
    } catch (e) {
      alert('导入失败：文件格式错误！');
      console.error(e);
    }
  };
  reader.readAsText(file);
}

function saveHistory() {
  // 1. 保存所有框的信息（含唯一ID）
  const boxes = Array.from(document.querySelectorAll('.box')).map(b => ({
    left: b.style.left,
    top: b.style.top,
    content: b.innerHTML,
    id: b.dataset.id,
    fontSize: b.style.fontSize,
    fontWeight: b.style.fontWeight,
    color: b.style.color
  }));
  // 2. 保存连线（转为ID，而非DOM引用）
  const savedConnections = connections.map(c => ({
    fromId: c.fromId,
    toId: c.toId
  }));
  history.push({ boxes, connections: savedConnections });
  if (history.length > 30) history.shift();
}

function undo() {
  if (history.length === 0) { alert('无操作可撤销'); return; }
  const last = history.pop();
  
  // 清空现有框和连线
  document.querySelectorAll('.box').forEach(b => b.remove());
  connections = [];
  
  // 重建框
  const boxMap = new Map(); // 映射 ID -> 新DOM节点
  last.boxes.forEach(data => {
    const box = document.createElement('div');
    box.className = 'box';
    box.contentEditable = true;
    box.style.left = data.left;
    box.style.top = data.top;
    box.innerHTML = data.content;
    box.style.fontSize = data.fontSize || '15px';
    box.style.fontWeight = data.fontWeight || '400';
    box.style.color = data.color || '#333';
    box.dataset.id = data.id;
    viewport.appendChild(box);
    makeDraggable(box);
    boxMap.set(data.id, box);
  });
  
  // 重建连线（根据ID关联新DOM节点）
  last.connections.forEach(c => {
    const fromBox = boxMap.get(c.fromId);
    const toBox = boxMap.get(c.toId);
    if (fromBox && toBox) {
      connections.push({ fromId: c.fromId, toId: c.toId });
    }
  });
  
  redrawAllConnections();
  // 撤销后同步保存到本地
  saveToLocalStorage();
}

  lastX = e.clientX;
  lastY = e.clientY;

  redrawAllConnections();
});

document.addEventListener('wheel', e => {
  e.preventDefault();
  const speed = 0.1;
  const mx = e.clientX - offsetX;
  const my = e.clientY - offsetY;
  const oldScale = scale;
  scale += e.deltaY < 0 ? speed : -speed;
  scale = Math.max(0.1, Math.min(scale, 10));
  offsetX = e.clientX - mx * (scale / oldScale);
  offsetY = e.clientY - my * (scale / oldScale);
  updateViewport();
  redrawAllConnections();
  // 缩放后保存
  saveToLocalStorage();
});

function updateViewport() {
  viewport.style.transform = `translate(${offsetX}px, ${offsetY}px) scale(${scale})`;
}

function makeDraggable(el) {
  let dragging = false;
  let sx, sy, l, t;

  el.addEventListener('mousedown', e => {
    e.stopPropagation();
    dragging = true;
    sx = e.clientX;
    sy = e.clientY;
    l = parseFloat(el.style.left) || 0;
    t = parseFloat(el.style.top) || 0;
    el.style.zIndex = 999;
    saveHistory(); // 拖动前保存历史
  });

  document.addEventListener('mousemove', e => {
    if (!dragging) return;
    const dx = (e.clientX - sx) / scale;
    const dy = (e.clientY - sy) / scale;
    let nl = l + dx;
    let nt = t + dy;

    // 磁吸逻辑
    document.querySelectorAll('.box').forEach(b => {
      if (b === el) return;
      const bl = parseFloat(b.style.left)||0;
      const bt = parseFloat(b.style.top)||0;
      const br = bl + b.offsetWidth;
      const bb = bt + b.offsetHeight;
      const w = el.offsetWidth;
      const h = el.offsetHeight;
      if (Math.abs(nl - bl) < MAGNET_THRESHOLD) nl = bl;
      if (Math.abs(nl + w - br) < MAGNET_THRESHOLD) nl = br - w;
      if (Math.abs(nt - bt) < MAGNET_THRESHOLD) nt = bt;
      if (Math.abs(nt + h - bb) < MAGNET_THRESHOLD) nt = bb - h;
    });

    el.style.left = nl + 'px';
    el.style.top = nt + 'px';
    redrawAllConnections();
  });

  document.addEventListener('mouseup', () => {
    dragging = false;
    el.style.zIndex = '';
    // 拖动结束后保存
    saveToLocalStorage();
  });
}

function redrawAllConnections() {
  connectionLayer.innerHTML = '';
  // 遍历连线（根据ID找DOM节点）
  connections.forEach(c => {
    const from = document.querySelector(`.box[data-id="${c.fromId}"]`);
    const to = document.querySelector(`.box[data-id="${c.toId}"]`);
    if (from && to) {
      drawPath(from, to);
    }
  });
}

function getEdgePoint(from, to) {
  const fx = parseFloat(from.style.left)||0;
  const fy = parseFloat(from.style.top)||0;
  const fw = from.offsetWidth;
  const fh = from.offsetHeight;
  const fcx = fx + fw/2;
  const fcy = fy + fh/2;
  const tcx = parseFloat(to.style.left) + to.offsetWidth/2;
  const tcy = parseFloat(to.style.top) + to.offsetHeight/2;
  const dx = tcx - fcx;
  const dy = tcy - fcy;
  if (Math.abs(dx) > Math.abs(dy)) {
    return dx > 0 ? {x:fx+fw, y:fcy} : {x:fx, y:fcy};
  } else {
    return dy > 0 ? {x:fcx, y:fy+fh} : {x:fcx, y:fy};
  }
}

function drawPath(from, to) {
  const A = getEdgePoint(from, to);
  const B = getEdgePoint(to, from);
  const ax = offsetX + A.x * scale;
  const ay = offsetY + A.y * scale;
  const bx = offsetX + B.x * scale;
  const by = offsetY + B.y * scale;
  const path = `M ${ax} ${ay} L ${bx} ${by}`;
  const p = document.createElementNS('http://www.w3.org/2000/svg', 'path');
  p.setAttribute('d', path);
  p.setAttribute('class', 'connection');
  connectionLayer.appendChild(p);
}
